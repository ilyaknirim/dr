📏 ДАТЧИК РАССТОЯНИЯ HC-SR04 · 7 ИНЖЕНЕРНЫХ ЭКСПЕРИМЕНТОВ

Принцип работы: датчик посылает ультразвуковой импульс (40 кГц) и засекает время возвращения эха. Расстояние = (время * скорость звука) / 2. Скорость звука в воздухе ≈ 343 м/с (зависит от температуры).

📐 Расстояние (см) = (длительность импульса в микросекундах) / 58
🔌 БАЗОВОЕ ПОДКЛЮЧЕНИЕ (все эксперименты)
VCC → 5V (питание)
GND → GND (земля)
Trig → D9 (пусковой сигнал)
Echo → D10 (вход эха)
⚠️ ВАЖНО: Выход Echo даёт 5В, что безопасно для Arduino. Но если используете длинные провода, добавьте резистор 1 кОм между Echo и землёй для помехозащиты.

// ЭКСПЕРИМЕНТ 1: Базовое измерение расстояния // Формула: расстояние (см) = время (мкс) / 58 #define TRIG_PIN 9 #define ECHO_PIN 10 void setup() { pinMode(TRIG_PIN, OUTPUT); pinMode(ECHO_PIN, INPUT); Serial.begin(9600); Serial.println("=== HC-SR04: ИЗМЕРЕНИЕ РАССТОЯНИЯ ==="); } void loop() { // Посылаем короткий импульс 10 мкс digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2); digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10); digitalWrite(TRIG_PIN, LOW); // Измеряем длительность эхо-импульса (в микросекундах) long duration = pulseIn(ECHO_PIN, HIGH); // Конвертируем в расстояние (см) int distance = duration / 58; // 58 = 2 * 10000 / 343 (грубо) Serial.print("Время: "); Serial.print(duration); Serial.print(" мкс | Расстояние: "); Serial.print(distance); Serial.println(" см"); delay(500); }
🔬 ФАКТ О КОДЕ: pulseIn() ждёт изменения сигнала и засекает время с точностью до микросекунды. Это аппаратная функция — она не использует прерывания и может блокировать программу на время ожидания (до 1 секунды). Для движущихся объектов лучше использовать прерывания, но pulseIn() проще для старта .
📊 ФИЗИКА ПРОЦЕССА: Звук проходит расстояние до объекта и обратно. Поэтому время делим на 2. Скорость звука зависит от температуры: v = 331 + 0.6 * T(°C) м/с. При 20°C это 343 м/с. Коэффициент 58 получен как: 10000 / (343/2) ≈ 58.3 .

// ЭКСПЕРИМЕНТ 2: Светофор расстояния // Зелёный: далеко (> 100 см) // Жёлтый: близко (30-100 см) // Красный: опасно близко (< 30 см) #define TRIG 9 #define ECHO 10 #define GREEN 11 #define YELLOW 12 #define RED 13 void setup() { pinMode(TRIG, OUTPUT); pinMode(ECHO, INPUT); pinMode(GREEN, OUTPUT); pinMode(YELLOW, OUTPUT); pinMode(RED, OUTPUT); Serial.begin(9600); } void loop() { // Измеряем расстояние digitalWrite(TRIG, LOW); delayMicroseconds(2); digitalWrite(TRIG, HIGH); delayMicroseconds(10); digitalWrite(TRIG, LOW); long duration = pulseIn(ECHO, HIGH); int distance = duration / 58; // Управление светодиодами if (distance > 100) { digitalWrite(GREEN, HIGH); digitalWrite(YELLOW, LOW); digitalWrite(RED, LOW); Serial.println("🟢 ДАЛЕКО"); } else if (distance > 30) { digitalWrite(GREEN, LOW); digitalWrite(YELLOW, HIGH); digitalWrite(RED, LOW); Serial.println("🟡 БЛИЗКО"); } else { digitalWrite(GREEN, LOW); digitalWrite(YELLOW, LOW); digitalWrite(RED, HIGH); Serial.println("🔴 ОПАСНО!"); } delay(200); }
🚦 ПРИМЕНЕНИЕ: Точно так работают парктроники в автомобилях. Разница только в количестве датчиков (обычно 4-8) и звуковой индикации. Частота писка увеличивается при приближении к препятствию.

// ЭКСПЕРИМЕНТ 3: Звуковой парктроник // Чем ближе объект, тем чаще писк #define TRIG 9 #define ECHO 10 #define BUZZER 8 void setup() { pinMode(TRIG, OUTPUT); pinMode(ECHO, INPUT); pinMode(BUZZER, OUTPUT); Serial.begin(9600); } void loop() { // Измерение digitalWrite(TRIG, LOW); delayMicroseconds(2); digitalWrite(TRIG, HIGH); delayMicroseconds(10); digitalWrite(TRIG, LOW); long duration = pulseIn(ECHO, HIGH); int distance = duration / 58; // Ограничиваем до 200 см (дальше не интересно) if (distance > 200) distance = 200; if (distance < 100) { // Пищим с частотой, обратной расстоянию int beepDelay = map(distance, 10, 100, 50, 500); beepDelay = constrain(beepDelay, 50, 500); tone(BUZZER, 2000, 50); // Частота 2 кГц, длительность 50 мс delay(beepDelay); } else { delay(100); } Serial.print("Расстояние: "); Serial.print(distance); Serial.print(" см | Задержка: "); Serial.println(map(distance, 10, 100, 50, 500)); }
🔊 ФАКТ О TONE(): Функция tone(pin, frequency, duration) генерирует меандр заданной частоты на пине. Важно: на пинах 3 и 11 используется тот же таймер, что и для analogWrite() на этих пинах — они не могут работать одновременно .

// ЭКСПЕРИМЕНТ 4: Ультразвуковой радар // Сканирует пространство от 0 до 180 градусов #include <Servo.h> #define TRIG 9 #define ECHO 10 #define SERVO_PIN 11 Servo myServo; int angle; void setup() { pinMode(TRIG, OUTPUT); pinMode(ECHO, INPUT); myServo.attach(SERVO_PIN); Serial.begin(9600); Serial.println("Angle,Distance"); } void loop() { // Сканируем слева направо for (angle = 15; angle <= 165; angle++) { myServo.write(angle); delay(30); // Ждём, пока сервопривод повернётся int distance = getDistance(); // Выводим в формате для Processing (CSV) Serial.print(angle); Serial.print(","); Serial.println(distance); } // Сканируем справа налево for (angle = 165; angle >= 15; angle--) { myServo.write(angle); delay(30); int distance = getDistance(); Serial.print(angle); Serial.print(","); Serial.println(distance); } } int getDistance() { digitalWrite(TRIG, LOW); delayMicroseconds(2); digitalWrite(TRIG, HIGH); delayMicroseconds(10); digitalWrite(TRIG, LOW); long duration = pulseIn(ECHO, HIGH, 30000); // Таймаут 30 мс if (duration == 0) return 400; // Нет объекта int distance = duration / 58; return distance; }
🛰️ ФАКТ О СКАНИРОВАНИИ: Данные из этого эксперимента можно визуализировать в Processing (см. официальный пример "Radar"). В реальных радарах (например, в автомобилях) используется несколько лучей или фазированная решётка, но принцип тот же — времяпролётный метод .

// ЭКСПЕРИМЕНТ 5: Вывод расстояния на LCD // + графическая шкала из символов █ #include <Wire.h> #include <LiquidCrystal_I2C.h> #define TRIG 9 #define ECHO 10 LiquidCrystal_I2C lcd(0x27, 16, 2); void setup() { pinMode(TRIG, OUTPUT); pinMode(ECHO, INPUT); lcd.init(); lcd.backlight(); lcd.setCursor(0, 0); lcd.print("Distance:"); Serial.begin(9600); } void loop() { // Измерение digitalWrite(TRIG, LOW); delayMicroseconds(2); digitalWrite(TRIG, HIGH); delayMicroseconds(10); digitalWrite(TRIG, LOW); long duration = pulseIn(ECHO, HIGH, 30000); int distance = duration / 58; if (distance > 400) distance = 400; // Вывод на LCD (первая строка) lcd.setCursor(0, 0); lcd.print("Distance: "); lcd.print(distance); lcd.print("cm "); // Графическая шкала (вторая строка) lcd.setCursor(0, 1); int bars = map(distance, 0, 200, 16, 0); bars = constrain(bars, 0, 16); for (int i = 0; i < bars; i++) lcd.print("█"); for (int i = bars; i < 16; i++) lcd.print("░"); Serial.print("Расстояние: "); Serial.println(distance); delay(100); }

📌 ЭКСПЕРИМЕНТ 6 · ИЗМЕРИТЕЛЬ ОБЪЁМА

Монтаж: Закрепите датчик над ёмкостью так, чтобы он смотрел вертикально вниз на поверхность воды.

// ЭКСПЕРИМЕНТ 6: Измерение объёма жидкости // Датчик сверху, измеряем расстояние до поверхности #define TRIG 9 #define ECHO 10 // Константы вашей ёмкости const float TANK_HEIGHT = 30.0; // Высота ёмкости (см) const float TANK_AREA = 200.0; // Площадь основания (см²) const float SENSOR_OFFSET = 5.0; // Расстояние от датчика до края ёмкости (см) void setup() { pinMode(TRIG, OUTPUT); pinMode(ECHO, INPUT); Serial.begin(9600); Serial.println("=== ИЗМЕРИТЕЛЬ ОБЪЁМА ==="); } void loop() { // Измеряем расстояние до воды digitalWrite(TRIG, LOW); delayMicroseconds(2); digitalWrite(TRIG, HIGH); delayMicroseconds(10); digitalWrite(TRIG, LOW); long duration = pulseIn(ECHO, HIGH, 30000); float distanceToWater = duration / 58.0; // см // Вычисляем высоту воды float waterHeight = TANK_HEIGHT - (distanceToWater - SENSOR_OFFSET); if (waterHeight < 0) waterHeight = 0; if (waterHeight > TANK_HEIGHT) waterHeight = TANK_HEIGHT; // Объём = площадь * высота float volume = TANK_AREA * waterHeight; // см³ = мл Serial.print("Расстояние до воды: "); Serial.print(distanceToWater); Serial.print(" см | Высота воды: "); Serial.print(waterHeight); Serial.print(" см | Объём: "); Serial.print(volume); Serial.println(" мл"); delay(1000); }
💧 ПРИМЕНЕНИЕ: Этот метод используется в промышленных резервуарах, но с ультразвуковыми уровнемерами (например, Siemens SITRANS). Точность зависит от формы ёмкости: для цилиндра формула сложнее (V = πr²h) .

// ЭКСПЕРИМЕНТ 7: Робот, объезжающий препятствия #define TRIG 9 #define ECHO 10 // Пины моторов (L293D) #define EN1 5 // Enable левого мотора #define IN1 6 // Input 1 #define IN2 7 // Input 2 #define IN3 8 // Input 3 #define IN4 11 // Input 4 #define EN2 12 // Enable правого мотора int speed = 150; // Скорость 0-255 void setup() { pinMode(TRIG, OUTPUT); pinMode(ECHO, INPUT); pinMode(EN1, OUTPUT); pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT); pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT); pinMode(EN2, OUTPUT); Serial.begin(9600); } void loop() { int distance = getDistance(); Serial.print("Расстояние: "); Serial.println(distance); if (distance > 30) { // Свободно - едем вперёд forward(); } else { // Препятствие - поворачиваем stop(); delay(300); backward(); delay(300); stop(); turnRight(); delay(400); } } int getDistance() { digitalWrite(TRIG, LOW); delayMicroseconds(2); digitalWrite(TRIG, HIGH); delayMicroseconds(10); digitalWrite(TRIG, LOW); long duration = pulseIn(ECHO, HIGH, 30000); return duration / 58; } void forward() { digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW); digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW); analogWrite(EN1, speed); analogWrite(EN2, speed); } void backward() { digitalWrite(IN1, LOW); digitalWrite(IN2, HIGH); digitalWrite(IN3, LOW); digitalWrite(IN4, HIGH); analogWrite(EN1, speed); analogWrite(EN2, speed); } void turnRight() { digitalWrite(IN1, LOW); digitalWrite(IN2, HIGH); digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW); analogWrite(EN1, speed); analogWrite(EN2, speed); } void stop() { digitalWrite(IN1, LOW); digitalWrite(IN2, LOW); digitalWrite(IN3, LOW); digitalWrite(IN4, LOW); analogWrite(EN1, 0); analogWrite(EN2, 0); }
🤖 ФАКТ О РОБОТАХ: Алгоритм "избегание препятствий" — основа поведения многих роботов (пылесосы Roomba, газонокосилки). В реальных устройствах используют несколько датчиков и более сложную логику (например, алгоритм Bug или потенциалы поля) .
